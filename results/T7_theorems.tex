% Theorem A (trajectory-tight guarantee)
% Status: lower bound [HAND-PROOF-UNREVIEWED] (classical (1/eta)-approximate greedy
%         analysis; Goundan--Schulz 2007 [CITATION-NEEDS-VERIFICATION]);
%         tightness instance [VERIFIED-SYMBOLIC] for GENERAL K

%         eta_u = ahat, eta_o = aK/(K-1), path error = ahat, ratio = L_K(ahat);
%         the only hand steps are a one-line induction that greedy stays at
%         y = 0, whose per-step tie identity IS symbolically verified, and the
%         faithfulness of the p = a^x parametrization, with a in (0,1) verified)
%         plus [VERIFIED-LP] full lattice K <= 6 (code/check_explicit_instance.py).
\begin{theorem}[Trajectory-tight guarantee]\label{thm:trajectory}
Let $f$ be monotone submodular with $f(\emptyset)=0$, and let
$S^0=\emptyset,S^1,\dots,S^K$ be the states of single-step predictive greedy run
with predictor $\tilde f$. If the path error satisfies
$\eta^{\mathrm{path}} \le \hat a$, then
\[
  f(S^K) \;\ge\; \Bigl(1-\bigl(1-\tfrac{1}{\hat a K}\bigr)^{K}\Bigr)\, f(O^\ast)
  \;=\; L_K(\hat a)\, f(O^\ast).
\]
Moreover, for every $K\ge 2$ and every $\hat a>1$ there is an instance
$(f,\tilde f)$ on $|N|=2K$ elements with $\eta^{\mathrm{path}}=\hat a$ exactly,
on which greedy (with adversarial tie breaking) returns a set of value exactly
$L_K(\hat a)\, f(O^\ast)$. Hence the bound is tight for every $K$, not only in
the limit.
\end{theorem}

% Lemma B (consistency lemma)
% Status: [HAND-PROOF-UNREVIEWED] (RESEARCH_STATE.md R3; proof is elementary,
%         uses only single-element error bounds).
\begin{lemma}[Consistency]\label{lem:consistency}
Let $f$ be monotone, $S\subseteq N$, $e,e'\notin S$, and suppose
$\tilde d_e(S)\ge \tilde d_{e'}(S)$. Then
\[
 \text{(i)}\quad d_e(S\cup\{e'\}) \;\ge\; \frac{1}{\eta}\, d_{e'}(S\cup\{e\}),
 \qquad
 \text{(ii)}\quad \Bigl(1-\frac{1}{\eta}\Bigr) d_{e'}(S\cup\{e\})
   \;\ge\; d_{e'}(S) - d_{e}(S).
\]
\end{lemma}

\begin{theorem}[Exact ratio for $K=2$]\label{thm:k2}
For every $\eta\ge 1$,
\[
  \rho_2(\eta) \;=\; \min\Bigl\{\frac{1}{\eta},\; \frac{3}{2(\eta+1)}\Bigr\}.
\]
\end{theorem}

% Theorem D (1/eta impossibility, with matching exhaustive search)
% Status: [HAND-PROOF-UNREVIEWED] (RESEARCH_STATE.md R2; elementary).
\begin{theorem}[Ceiling at $1/\eta$]\label{thm:ceiling}
For every deterministic algorithm with arbitrary (even exponential) query access
to $\tilde f$ and every $\eta_u,\eta_o \ge 1$ with $n\ge 2K$, there is a pair
$(f,\tilde f)$ with all-pairs errors exactly $(\eta_u,\eta_o)$ on which the
algorithm's output $S$ satisfies $f(S)\le f(O^\ast)/\eta$. Randomized
algorithms lose at most the additive term $K/n$:
$\mathbb{E}[f(S)] \le \bigl((1-K/n)/\eta + K/n\bigr) f(O^\ast)$.
Conversely, exhaustive search over all $K$-subsets of the predicted values
achieves $f(\hat S) \ge f(O^\ast)/\eta$ for every instance.
\end{theorem}

% Corollary E (weak submodularity)
% Status: [HAND-PROOF-UNREVIEWED]; submodularity-ratio framework of
%         Das--Kempe 2011 [CITATION-NEEDS-VERIFICATION].
\begin{corollary}[Weakly submodular objectives]\label{cor:weak}
If $f$ is monotone with submodularity ratio $\gamma\in(0,1]$ (Das--Kempe), then
single-step predictive greedy with path error $\eta^{\mathrm{path}}\le\eta$
returns $S^K$ with
\[
  f(S^K) \;\ge\; \Bigl(1-\bigl(1-\tfrac{\gamma}{\eta K}\bigr)^{K}\Bigr) f(O^\ast).
\]
\end{corollary}
