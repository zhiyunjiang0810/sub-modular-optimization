\section{Setting and results}\label{sec:results}

% ---------------------------------------------------------------- block 1 --
\subsection{Model and notation}\label{sec:model}

A ground set $N$ with $|N|=n$, a monotone submodular $f:2^{N}\to\mathbb
R_{\ge0}$ with $f(\emptyset)=0$, and a cardinality budget $K$ are fixed.  The
algorithm cannot evaluate $f$; it can only query a predictor
$\tilde f:2^{N}\to\mathbb R$ with $\tilde f(\emptyset)=0$.  Marginal gains are
written $d_e(S)=f(S\cup\{e\})-f(S)$ and
$\tilde d_e(S)=\tilde f(S\cup\{e\})-\tilde f(S)$.

\begin{definition}[Prediction error]\label{def:eta}
For $\eta_u,\eta_o\ge1$ the predictor $\tilde f$ has (single-element,
multiplicative) error at most $(\eta_u,\eta_o)$ if for every $S\subseteq N$
and $e\notin S$,
\[
  d_e(S)/\eta_u\;\le\;\tilde d_e(S)\;\le\;\eta_o\,d_e(S).
\]
In particular $d_e(S)=0$ forces $\tilde d_e(S)=0$, and all predicted gains are
nonnegative.  The scalar error is $\eta=\eta_u\eta_o$.
\end{definition}

Approximation ratios are stated as $\alpha\in(0,1]$ with
$F^{\mathrm{ALG}}\ge\alpha\,F^{\mathrm{OPT}}$; larger is better.
% Erratum guard: the submitted draft stated the ratio in the opposite
% direction in Section 1.1; fixed here (RESEARCH_STATE, known errors).

Predictive greedy is the single-step greedy run on $\tilde f$: for
$t=0,\dots,K-1$ it adds an element maximizing $\tilde d_e(S^{t})$, with ties
broken adversarially in all worst-case statements.  Its exact worst-case ratio
at error level $\eta$ is written $\rho_K(\eta)$.  Two run-dependent error
measures refine $\eta$:
\begin{definition}[Selection error]\label{def:etasel}
For an execution of predictive greedy with states
$S^{0},\dots,S^{K-1}$ and picks $e_{0},\dots,e_{K-1}$, the selection error is
\[
  \etasel\;=\;\max\Bigl\{\,\max_{e\notin S^{t}} d_{e}(S^{t})\,/\,d_{e_t}(S^{t})
  \;:\;t<K,\ d_{e_t}(S^{t})>0\Bigr\}\;\ge\;1 ,
\]
the largest factor by which a step of the run fell short of the best true
gain, over the steps with positive chosen gain.  In the terminology of
Goundan and Schulz, a run with selection error $\etasel$ is an execution of
greedy with an $\alpha$-approximate incremental oracle with
$\alpha=\etasel$ \citep{goundan2007revisiting}.
% Status of the attribution: F6 audit item 4 (their alpha >= 1 satisfies
% alpha rho_{e_t} >= max_e rho_e, so alpha = etasel; the earlier 1/etasel
% reading was reversed and is corrected here).  The notation \etasel is ours.
% Their Theorem 1 (p. 7) is verbatim the L_K bound; Theorem
% \ref{thm:guarantee} below is that theorem instantiated and is attributed
% accordingly, not claimed as new.
\end{definition}
The trajectory error restricts Definition~\ref{def:eta} to the states of the
run; it is used in a single remark below and written $\etatr$ when a symbol is
needed.  The chain $\etasel\le\etatr\le\eta$ holds for every run.

\subsection{An error bound is necessary}\label{sec:necessity}

\begin{proposition}[No bound, no guarantee]\label{prop:necessity}
If no upper bound on $\eta$ is assumed, then for every algorithm with
arbitrary query access to $\tilde f$ and every $n\ge2K$ there are pairs
$(f,\tilde f)$ on which the output $T$ satisfies
$f(T)\le\tfrac{K}{n-K}\,f(O^{\ast})$.  Consequently no constant worst-case
ratio is achievable without an error assumption.
\end{proposition}

\subsection{The guarantee, stated for the selection error}\label{sec:guarantee}

\begin{theorem}[Guarantee of predictive greedy]\label{thm:guarantee}
Let $f$ be monotone submodular with $f(\emptyset)=0$ and let $T=S^{K}$ be the
output of a run of predictive greedy whose selection error is $\etasel$.
Then, with $L_K(x)=1-(1-\tfrac1{xK})^{K}$,
\[
  f(T)\;\ge\;L_K(\etasel)\,f(O^{\ast})
  \;\ge\;\bigl(1-e^{-1/\etasel}\bigr)f(O^{\ast}).
\]
The same bound therefore holds with $\etasel$ replaced by $\etatr$ or by
$\eta$, and the three bounds are ordered
$L_K(\etasel)\ge L_K(\etatr)\ge L_K(\eta)$.
\end{theorem}

\begin{remark}\label{rem:etasel-measurable}
$\etasel$ is computable from the run when $f$ becomes observable in
hindsight, which makes Theorem~\ref{thm:guarantee} a certificate: the
measured $\etasel$ of a finished run certifies the printed bound for that
run.  The experiments of Section~\ref{sec:experiments} report measured
$\etasel$ and the resulting certified bounds.
\end{remark}

\subsection{Per-$K$ tightness}\label{sec:tightness}

\begin{theorem}[Tightness for every $K$]\label{thm:tightness}
For every $K\ge2$ and $\hat a>1$ there is an instance $(f,\tilde f)$ on
$2K$ elements with $\etasel=\etatr=\hat a$ on the adversarial-tie run of
predictive greedy, whose output value is exactly $L_K(\hat a)\,f(O^{\ast})$.
Hence Theorem~\ref{thm:guarantee} is an equality on this family for every
$K$, not only in the limit.
\end{theorem}

\subsection{The coherence lemma}\label{sec:coherence}

\begin{lemma}[Coherence]\label{lem:coherence}
Let $f$ be monotone, $S\subseteq N$, $e,e'\notin S$, and suppose
$\tilde d_{e}(S)\ge\tilde d_{e'}(S)$.  Then
\[
  \text{(i)}\;\; d_{e}(S\cup\{e'\})\ge\tfrac1\eta\,d_{e'}(S\cup\{e\}),
  \qquad
  \text{(ii)}\;\;\bigl(1-\tfrac1\eta\bigr)\,d_{e'}(S\cup\{e\})
  \ge d_{e'}(S)-d_{e}(S).
\]
\end{lemma}

\subsection{The exact worst case}\label{sec:exact}

For $K\ge2$, $\eta\ge1$, let $k_1=(K-1)\eta+1$, $q=(K-1)\eta/k_1$, and for
$0\le j\le K-1$
\[
  V_j(\eta)\;=\;1-q^{\,j}\Bigl(1-\frac{K-j}{K\eta}\Bigr).
\]

\begin{theorem}[Exact worst-case ratio of predictive greedy]\label{thm:exact}
For every $K\ge2$ and $\eta\ge1$, under adversarial tie breaking,
\[
  \rho_K(\eta)\;=\;\min_{0\le j\le K-1}V_j(\eta),
\]
and the minimum is attained by $V_j$ on the segment
$\eta\in[K-j,K-j+1]$ (with $V_0=1/\eta$ on $[K,\infty)$), so the breakpoints
are the integers $2,\dots,K$.  In particular
$\rho_K(\eta)=1/\eta$ exactly when $\eta\ge K$, and for $K\in\{2,3,4\}$ the
closed forms are
$\rho_2=\min\{\tfrac1\eta,\tfrac{3}{2(\eta+1)}\}$,
$\rho_3=\tfrac{16\eta+3}{3(2\eta+1)^2}$, $\tfrac{7}{3(2\eta+1)}$,
$\tfrac1\eta$ on $[1,2],[2,3],[3,\infty)$, and
$\rho_4=\tfrac{135\eta^2+36\eta+4}{4(3\eta+1)^3}$,
$\tfrac{21\eta+2}{2(3\eta+1)^2}$, $\tfrac{13}{4(3\eta+1)}$, $\tfrac1\eta$ on
$[1,2],[2,3],[3,4],[4,\infty)$.
\end{theorem}

\begin{remark}\label{rem:exact-gap}
$L_K$ and $\rho_K$ differ by $O(1/K)$ for fixed $\eta$, and the previously
used explicit family attains only
$U_K(\eta)=1-(1-\tfrac{1}{\eta(K-1)+1})^{K}$, which is strictly above
$V_{K-1}$ for every $\eta>1$.
% [VERIFIED-SYMBOLIC]: U_K - V_{K-1} = q^{K-1}(eta-1)/(K eta k_1) > 0,
% results/N1_dual_certificate.py part G.
The predictor $\tilde f$ of the attaining instances is monotone but not
submodular.  If the model additionally requires $\tilde f$ to be submodular,
the worst case changes: for $\eta<K-1$ the exact value strictly increases
(for example from $9/16$ to $19/33$ at $K=3$, $\eta=1.5$), the family above
becomes inadmissible, and $U_K$ is no longer an upper bound; LP evidence
supports the closed form $\min_m W_m$ with
$W_m=\tfrac{K-mr^{m}}{K(1+(\eta-1)r^{m})}$, $r=1-\tfrac1K$, whose limit in
$K$ exceeds $1-e^{-1/\eta}$ by roughly ten percent at $\eta=2$.

\subsection{The $1/\eta$ ceiling}\label{sec:ceiling}

\begin{theorem}[Ceiling at $1/\eta$]\label{thm:ceiling}
For every deterministic algorithm with arbitrary query access to $\tilde f$,
every $\eta_u,\eta_o\ge1$, and every $n\ge2K$, there is a pair
$(f,\tilde f)$ with error exactly $(\eta_u,\eta_o)$ on which the output
$T$ satisfies $f(T)\le f(O^{\ast})/\eta$.  For randomized algorithms the
bound becomes $(1-\tfrac Kn)\tfrac1\eta+\tfrac Kn$ in expectation.
Conversely, exhaustive search over all $K$-subsets of predicted values
achieves $f(\hat S)\ge f(O^{\ast})/\eta$ on every instance.
\end{theorem}

\subsection{Asymptotics}\label{sec:asymptotics}

\begin{corollary}[Limit in $K$]\label{cor:limit}
For fixed $\eta\ge1$, both $L_K(\eta)$ and $\rho_K(\eta)$ converge to
$1-e^{-1/\eta}$ as $K\to\infty$, monotonically within each segment.
\end{corollary}

\subsection{Hardness for bounded-query algorithms}\label{sec:hardness}

The ceiling of Theorem~\ref{thm:ceiling} does not restrict the query budget.
The final result shows that with polynomially many small queries no algorithm
can beat the greedy guarantee by more than a vanishing margin.  For
$\theta\ge1$ let $a_\theta=1-\tfrac1{\theta K}$,
$1+\delta(\theta)=\max\{a_\theta^{\tau}\tfrac{K}{K-\tau},
a_\theta^{1-\tau}\}^{2}$ and $\Phi(\theta)=\theta(1+\delta(\theta))$.

\begin{theorem}[Bounded-query hardness, deterministic]\label{thm:hardness}
Fix $c\ge0$, put $\tau=c+1$, and let $K\ge2\tau$ and $\eta>1$ satisfy
$\Phi(1)\le\eta$ (sufficient: $K\ge\tau(1+\tfrac{2}{\ln\eta})$), so that
$\hat\eta=\Phi^{-1}(\eta)\in[1,\eta]$ is well defined.  Let $n\ge4K^{c+2}$.
For every deterministic algorithm making at most $n^{c}$ queries to
$\tilde f$, each on a set of size at most $K$, there is an instance
$(f,\tilde f)$ with monotone submodular $f$ and single-element error exactly
$\eta$ on which the output $T$ (of size at most $K$) satisfies
\[
  f(T)\;\le\;L_K(\hat\eta)\,f(O^{\ast}).
\]
For randomized algorithms the same bound holds up to an additive
$\varepsilon_n=\tfrac Kn+\tfrac{K^{2c+4}}{(c+2)!\,n^{2}}$ in expectation.
Moreover $K\delta(\eta)\to\max\{2\tau(1-\tfrac1\eta),\tfrac{2(\tau-1)}\eta\}$
and $L_K(\hat\eta)\to1-e^{-1/\eta}$ as $K\to\infty$.
\end{theorem}

\begin{remark}\label{rem:hardness-pins}
Combined with Theorem~\ref{thm:guarantee}, the bounded-query optimum at error
level $\eta$ is pinned to $1-e^{-1/\eta}$ asymptotically in $K$: predictive
greedy achieves $L_K(\eta)\to1-e^{-1/\eta}$ with $K$ evaluations of
$\tilde f$ per step, and no algorithm within the stated query budget exceeds
$L_K(\hat\eta)\to1-e^{-1/\eta}$.  At finite $K$ a gap of order $c/K$ between
the two curves remains open.
\end{remark}
